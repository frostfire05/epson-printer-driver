Write an MASM program to transfer a block of 10 data bytes from one memory location, 3000H, to another memory location, 4000H. Before transfer, multiply 5, then add 10 to each element.

MASM Source Code
.MODEL SMALL
.STACK 100H

PRINT MACRO TXT              ; print a '$'-terminated string
    LEA DX, TXT
    MOV AH, 09H
    INT 21H
ENDM

.DATA
MSG1 DB 'Source block (2000:3000H) : $'
MSG2 DB 0DH,0AH,'Dest block   (2000:4000H) : $'
MSG3 DB 0DH,0AH,'Each element = (source x 5) + 10$'
CRLF DB 0DH,0AH,'$'

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX               ; DS -> messages
    MOV AX, 2000H
    MOV ES, AX               ; ES -> segment holding both blocks (offsets 3000H / 4000H)

    ; ---- fill source block 3000H with 1..10 ----
    MOV SI, 3000H            ; SI = source pointer
    MOV CX, 10               ; CX = element count
    MOV AL, 1
FILLLP:
    MOV ES:[SI], AL
    INC AL
    INC SI
    LOOP FILLLP

    ; ---- transfer: dest = source * 5 + 10 ----
    MOV SI, 3000H            ; SI = source
    MOV DI, 4000H            ; DI = destination
    MOV CX, 10
    MOV BL, 5                ; BL = multiplier
XFER:
    MOV AL, ES:[SI]          ; AL = source byte
    MUL BL                   ; AX = AL * 5
    ADD AL, 10               ; AL = AL + 10
    MOV ES:[DI], AL          ; store at destination
    INC SI
    INC DI
    LOOP XFER

    ; ---- display both blocks ----
    PRINT MSG1
    MOV SI, 3000H
    MOV CX, 10
    CALL SHOWBLK
    PRINT MSG2
    MOV SI, 4000H
    MOV CX, 10
    CALL SHOWBLK
    PRINT MSG3
    PRINT CRLF

    MOV AX, 4C00H            ; exit to DOS
    INT 21H
MAIN ENDP

; SHOWBLK: print CX bytes starting at ES:SI as decimal numbers
SHOWBLK PROC
SB1:
    MOV AL, ES:[SI]
    XOR AH, AH
    CALL PRINT_NUM
    MOV DL, ' '
    MOV AH, 02H
    INT 21H
    INC SI
    LOOP SB1
    RET
SHOWBLK ENDP

; PRINT_NUM: print unsigned number in AX (all registers preserved)
PRINT_NUM PROC
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX
    MOV BX, 10               ; divisor
    XOR CX, CX               ; digit counter
PN1:
    XOR DX, DX
    DIV BX                   ; AX = quotient, DX = remainder digit
    PUSH DX
    INC CX
    OR AX, AX
    JNZ PN1
PN2:
    POP DX
    ADD DL, '0'
    MOV AH, 02H
    INT 21H
    LOOP PN2
    POP DX
    POP CX
    POP BX
    POP AX
    RET
PRINT_NUM ENDP

END MAIN

Write an MASM program to store an array of 20 numbers in zig-zag (alternating less than and greater than) order in memory location 4000H.

MASM Source Code
.MODEL SMALL
.STACK 100H

PRINT MACRO TXT
    LEA DX, TXT
    MOV AH, 09H
    INT 21H
ENDM

.DATA
ARR  DB 4,3,7,8,6,2,1,9,12,5,15,11,10,20,18,14,19,13,17,16   ; 20 input numbers
MSG1 DB 'Original array (copied to 2000:4000H):',0DH,0AH,'$'
MSG2 DB 0DH,0AH,0DH,0AH,'Zig-zag array (a0 < a1 > a2 < a3 ...):',0DH,0AH,'$'
CRLF DB 0DH,0AH,'$'

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX               ; DS -> data segment
    MOV AX, 2000H
    MOV ES, AX               ; ES -> segment that holds offset 4000H

    ; ---- copy 20 numbers to ES:4000H ----
    LEA SI, ARR              ; SI = source pointer (DS)
    MOV DI, 4000H            ; DI = destination pointer (ES)
    MOV CX, 20               ; CX = count
CPYLP:
    MOV AL, [SI]
    MOV ES:[DI], AL
    INC SI
    INC DI
    LOOP CPYLP

    PRINT MSG1
    MOV SI, 4000H
    MOV CX, 20
    CALL SHOWBLK

    ; ---- zig-zag pass: even i needs a[i] < a[i+1], odd i needs a[i] > a[i+1] ----
    MOV SI, 4000H            ; SI = pointer to a[i]
    MOV CX, 19               ; 19 adjacent pairs
    MOV DL, 0                ; DL = 0 (even index) / 1 (odd index)
ZZ:
    MOV AL, ES:[SI]          ; AL = a[i]
    MOV AH, ES:[SI+1]        ; AH = a[i+1]
    CMP DL, 0
    JNE ODDIDX
    CMP AL, AH               ; even index: need AL <= AH
    JBE NOSWAP
    JMP DOSWAP
ODDIDX:
    CMP AL, AH               ; odd index: need AL >= AH
    JAE NOSWAP
DOSWAP:
    MOV ES:[SI], AH          ; swap the pair
    MOV ES:[SI+1], AL
NOSWAP:
    XOR DL, 1                ; toggle even/odd
    INC SI
    LOOP ZZ

    PRINT MSG2
    MOV SI, 4000H
    MOV CX, 20
    CALL SHOWBLK
    PRINT CRLF

    MOV AX, 4C00H
    INT 21H
MAIN ENDP

; SHOWBLK: print CX bytes starting at ES:SI as decimal numbers
SHOWBLK PROC
SB1:
    MOV AL, ES:[SI]
    XOR AH, AH
    CALL PRINT_NUM
    MOV DL, ' '
    MOV AH, 02H
    INT 21H
    INC SI
    LOOP SB1
    RET
SHOWBLK ENDP

; PRINT_NUM: print unsigned number in AX (all registers preserved)
PRINT_NUM PROC
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX
    MOV BX, 10
    XOR CX, CX
PN1:
    XOR DX, DX
    DIV BX
    PUSH DX
    INC CX
    OR AX, AX
    JNZ PN1
PN2:
    POP DX
    ADD DL, '0'
    MOV AH, 02H
    INT 21H
    LOOP PN2
    POP DX
    POP CX
    POP BX
    POP AX
    RET
PRINT_NUM ENDP

END MAIN

Write an MASM program to convert a hexadecimal number into its equivalent ASCII.

MASM Source Code
.MODEL SMALL
.STACK 100H

PRINT MACRO TXT              ; print a '$'-terminated string
    LEA DX, TXT
    MOV AH, 09H
    INT 21H
ENDM

PUTC MACRO CHR               ; print one character
    MOV DL, CHR
    MOV AH, 02H
    INT 21H
ENDM

.DATA
HEXNUM DW 2F5AH              ; 16-bit hexadecimal number to convert
BUF    DB 4 DUP(?),'$'       ; receives the 4 ASCII characters
MSG1   DB 'Hex number (HEXNUM)  = 2F5AH',0DH,0AH,'$'
MSG2   DB 'ASCII characters     : $'
MSG3   DB 0DH,0AH,'ASCII codes (hex)    : $'
CRLF   DB 0DH,0AH,'$'

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX

    MOV AX, HEXNUM           ; AX = number to convert
    LEA DI, BUF              ; DI = output buffer pointer
    MOV CH, 4                ; CH = 4 nibbles to process
CONVLP:
    MOV CL, 4
    ROL AX, CL               ; bring the next (highest) nibble into AL bits 3-0
    PUSH AX                  ; keep rotated value
    AND AL, 0FH              ; isolate the nibble
    CALL NIB2ASC             ; AL = ASCII character
    MOV [DI], AL             ; store in buffer
    INC DI
    POP AX
    DEC CH
    JNZ CONVLP

    PRINT MSG1
    PRINT MSG2
    PRINT BUF                ; show the converted ASCII string
    PRINT MSG3

    ; ---- show the ASCII code of every character in hex ----
    LEA SI, BUF              ; SI = pointer to the characters
    MOV CH, 4
SHOWC:
    MOV AL, [SI]             ; AL = ASCII code
    PUSH AX
    MOV CL, 4
    SHR AL, CL               ; high nibble of the code
    CALL NIB2ASC
    PUTC AL
    POP AX
    AND AL, 0FH              ; low nibble of the code
    CALL NIB2ASC
    PUTC AL
    PUTC 'H'
    PUTC ' '
    INC SI
    DEC CH
    JNZ SHOWC
    PRINT CRLF

    MOV AX, 4C00H
    INT 21H
MAIN ENDP

; NIB2ASC: AL = nibble (0-0FH)  ->  AL = ASCII '0'-'9' or 'A'-'F'
NIB2ASC PROC
    CMP AL, 9
    JBE DIGIT
    ADD AL, 7                ; skip the gap between '9' and 'A'
DIGIT:
    ADD AL, '0'
    RET
NIB2ASC ENDP

END MAIN

Write an MASM program to implement a basic calculator for arithmetic operations.

MASM Source Code
.MODEL SMALL
.STACK 100H

PRINT MACRO TXT
    LEA DX, TXT
    MOV AH, 09H
    INT 21H
ENDM

.DATA
P1      DB 'Enter first number  : $'
P_OP    DB 'Enter operator (+ - * /) : $'
P2      DB 'Enter second number : $'
RESMSG  DB 'Result = $'
REMMSG  DB '  remainder $'
ERRDZ   DB 'Error: division by zero!$'
ERROP   DB 'Error: invalid operator!$'
OVFMSG  DB 'Error: result exceeds 16 bits!$'
AGAINM  DB 'Calculate again (Y/N)? $'
CRLF    DB 0DH,0AH,'$'
NUM1    DW ?
NUM2    DW ?
OPR     DB ?

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX

START:
    PRINT P1
    CALL READ_NUM            ; AX = first operand
    MOV NUM1, AX
    PRINT CRLF
    PRINT P_OP
    MOV AH, 01H              ; read operator character
    INT 21H
    MOV OPR, AL
    PRINT CRLF
    PRINT P2
    CALL READ_NUM            ; AX = second operand
    MOV NUM2, AX
    PRINT CRLF
    PRINT RESMSG

    MOV AX, NUM1             ; AX = operand 1
    MOV BX, NUM2             ; BX = operand 2
    CMP OPR, '+'
    JE DO_ADD
    CMP OPR, '-'
    JE DO_SUB
    CMP OPR, '*'
    JE DO_MUL
    CMP OPR, '/'
    JE DO_DIV
    PRINT ERROP
    JMP ASKAGAIN

DO_ADD:
    ADD AX, BX
    CALL PRINT_NUM
    JMP ASKAGAIN

DO_SUB:
    SUB AX, BX
    JNB SUBPOS               ; no borrow: result is positive
    NEG AX                   ; negative result: take magnitude
    PUSH AX
    MOV DL, '-'
    MOV AH, 02H
    INT 21H                  ; print the minus sign
    POP AX
SUBPOS:
    CALL PRINT_NUM
    JMP ASKAGAIN

DO_MUL:
    MUL BX                   ; DX:AX = AX * BX
    OR DX, DX
    JZ MULOK
    PRINT OVFMSG
    JMP ASKAGAIN
MULOK:
    CALL PRINT_NUM
    JMP ASKAGAIN

DO_DIV:
    OR BX, BX
    JNZ DIVOK
    PRINT ERRDZ
    JMP ASKAGAIN
DIVOK:
    XOR DX, DX               ; DX:AX = dividend
    DIV BX                   ; AX = quotient, DX = remainder
    CALL PRINT_NUM
    PUSH DX
    PRINT REMMSG
    POP AX
    CALL PRINT_NUM           ; print remainder

ASKAGAIN:
    PRINT CRLF
    PRINT AGAINM
    MOV AH, 01H
    INT 21H
    MOV BL, AL               ; save the answer
    PRINT CRLF
    PRINT CRLF
    CMP BL, 'Y'
    JE RESTART
    CMP BL, 'y'
    JE RESTART
    MOV AX, 4C00H            ; exit to DOS
    INT 21H
RESTART:
    JMP START
MAIN ENDP

; READ_NUM: read an unsigned decimal number (ends with Enter); returns AX
READ_NUM PROC
    PUSH BX
    PUSH CX
    PUSH DX
    XOR BX, BX               ; BX = running value
RN1:
    MOV AH, 01H
    INT 21H
    CMP AL, 0DH              ; Enter ends the number
    JE RN2
    CMP AL, '0'
    JB RN1                   ; ignore non-digits
    CMP AL, '9'
    JA RN1
    SUB AL, '0'
    XOR AH, AH
    MOV CX, AX               ; CX = new digit
    MOV AX, BX
    MOV DX, 10
    MUL DX                   ; AX = value * 10
    ADD AX, CX               ; add digit
    MOV BX, AX
    JMP RN1
RN2:
    MOV AX, BX
    POP DX
    POP CX
    POP BX
    RET
READ_NUM ENDP

; PRINT_NUM: print unsigned number in AX (all registers preserved)
PRINT_NUM PROC
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX
    MOV BX, 10
    XOR CX, CX
PN1:
    XOR DX, DX
    DIV BX
    PUSH DX
    INC CX
    OR AX, AX
    JNZ PN1
PN2:
    POP DX
    ADD DL, '0'
    MOV AH, 02H
    INT 21H
    LOOP PN2
    POP DX
    POP CX
    POP BX
    POP AX
    RET
PRINT_NUM ENDP

END MAIN

Implement a library of string operations using assembly: (a) STRLEN: Return length of string, (b) STRCMP: Compare two strings, (c) STRREV: Reverse string in place.

MASM Source Code
.MODEL SMALL
.STACK 100H

PRINT MACRO TXT
    LEA DX, TXT
    MOV AH, 09H
    INT 21H
ENDM

.DATA
STR1   DB 'Assembly',0              ; all strings are NULL (0) terminated
STR2   DB 'Assembly',0
STR3   DB 'Assemble',0
STR4   DB 'System Programming',0
T_LEN  DB '(a) STRLEN',0DH,0AH,'$'
L1     DB '  Length of "Assembly"           = $'
L2     DB '  Length of "System Programming" = $'
T_CMP  DB '(b) STRCMP',0DH,0AH,'$'
C1     DB '  "Assembly" vs "Assembly" : $'
C2     DB '  "Assembly" vs "Assemble" : $'
C3     DB '  "Assemble" vs "Assembly" : $'
TEQ    DB 'EQUAL (AL = 00H)$'
TGT    DB 'STR1 GREATER (AL = 01H)$'
TLT    DB 'STR1 LESS (AL = FFH)$'
T_REV  DB '(c) STRREV',0DH,0AH,'$'
BEF    DB '  Before : $'
AFT    DB '  After  : $'
CRLF   DB 0DH,0AH,'$'

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX

    ; ---------- (a) STRLEN ----------
    PRINT T_LEN
    PRINT L1
    LEA SI, STR1             ; SI = string address
    CALL STRLEN              ; AX = length
    CALL PRINT_NUM
    PRINT CRLF
    PRINT L2
    LEA SI, STR4
    CALL STRLEN
    CALL PRINT_NUM
    PRINT CRLF

    ; ---------- (b) STRCMP ----------
    PRINT T_CMP
    PRINT C1
    LEA SI, STR1             ; SI = string 1
    LEA DI, STR2             ; DI = string 2
    CALL STRCMP              ; AL = 0 / 1 / FFH
    CALL SHOWCMP
    PRINT CRLF
    PRINT C2
    LEA SI, STR1
    LEA DI, STR3
    CALL STRCMP
    CALL SHOWCMP
    PRINT CRLF
    PRINT C3
    LEA SI, STR3
    LEA DI, STR1
    CALL STRCMP
    CALL SHOWCMP
    PRINT CRLF

    ; ---------- (c) STRREV ----------
    PRINT T_REV
    PRINT BEF
    LEA SI, STR4
    CALL PUTS
    PRINT CRLF
    LEA SI, STR4
    CALL STRREV              ; reverse in place
    PRINT AFT
    LEA SI, STR4
    CALL PUTS
    PRINT CRLF

    MOV AX, 4C00H
    INT 21H
MAIN ENDP

; STRLEN: SI = string address (NULL terminated); returns AX = length
STRLEN PROC
    PUSH SI
    XOR AX, AX               ; AX = counter
SL1:
    CMP BYTE PTR [SI], 0     ; end of string?
    JE SL2
    INC SI
    INC AX
    JMP SL1
SL2:
    POP SI
    RET
STRLEN ENDP

; STRCMP: SI = string 1, DI = string 2
; returns AL = 0 (equal), 1 (string1 > string2), 0FFH (string1 < string2)
STRCMP PROC
    PUSH SI
    PUSH DI
    PUSH BX
SC1:
    MOV AL, [SI]
    MOV BL, [DI]
    CMP AL, BL
    JNE SC2                  ; first differing character found
    CMP AL, 0
    JE SCEQ                  ; both ended together: equal
    INC SI
    INC DI
    JMP SC1
SC2:
    JA SCGT                  ; AL above BL: string1 greater
    MOV AL, 0FFH             ; otherwise string1 is less
    JMP SCEND
SCGT:
    MOV AL, 1
    JMP SCEND
SCEQ:
    MOV AL, 0
SCEND:
    POP BX
    POP DI
    POP SI
    RET
STRCMP ENDP

; STRREV: SI = string address; reverses the string in place (two-pointer swap)
STRREV PROC
    PUSH AX
    PUSH BX
    PUSH SI
    PUSH DI
    CALL STRLEN              ; AX = length
    CMP AX, 2
    JB SRDONE                ; nothing to reverse
    MOV DI, SI
    ADD DI, AX
    DEC DI                   ; DI = last character, SI = first character
SR1:
    CMP SI, DI
    JAE SRDONE               ; pointers met or crossed
    MOV AL, [SI]
    MOV BL, [DI]
    MOV [SI], BL             ; swap the two characters
    MOV [DI], AL
    INC SI
    DEC DI
    JMP SR1
SRDONE:
    POP DI
    POP SI
    POP BX
    POP AX
    RET
STRREV ENDP

; SHOWCMP: AL = STRCMP result; prints its meaning
SHOWCMP PROC
    CMP AL, 0
    JE SH_EQ
    CMP AL, 1
    JE SH_GT
    PRINT TLT
    RET
SH_GT:
    PRINT TGT
    RET
SH_EQ:
    PRINT TEQ
    RET
SHOWCMP ENDP

; PUTS: print NULL terminated string at SI
PUTS PROC
    PUSH AX
    PUSH DX
    PUSH SI
PU1:
    MOV DL, [SI]
    CMP DL, 0
    JE PU2
    MOV AH, 02H
    INT 21H
    INC SI
    JMP PU1
PU2:
    POP SI
    POP DX
    POP AX
    RET
PUTS ENDP

; PRINT_NUM: print unsigned number in AX (all registers preserved)
PRINT_NUM PROC
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX
    MOV BX, 10
    XOR CX, CX
PN1:
    XOR DX, DX
    DIV BX
    PUSH DX
    INC CX
    OR AX, AX
    JNZ PN1
PN2:
    POP DX
    ADD DL, '0'
    MOV AH, 02H
    INT 21H
    LOOP PN2
    POP DX
    POP CX
    POP BX
    POP AX
    RET
PRINT_NUM ENDP

END MAIN

Boot-level password verification system: Simulate a BIOS-like password verification system, where the steps would be: (a) accepting hidden password input (no echo), (b) comparing with a hardcoded value, (c) allowing up to 3 retries before locking, (d) using BIOS interrupts (INT 16h / INT 10h) for keyboard handling.

MASM Source Code
.MODEL SMALL
.STACK 100H

.DATA
PASSWD  DB 'MASM86',0                 ; hardcoded password (NULL terminated)
BUF     DB 20 DUP(0)                  ; buffer for the typed password
MAXLEN  EQU 19                        ; maximum characters accepted
TRIES   DB 3                          ; attempts left before lock
BANNER  DB '=== SIMPLE BIOS PASSWORD CHECK ===',0DH,0AH,0
PROMPT  DB 'Enter password: ',0
WRONG   DB 'Access denied. Attempts left: ',0
GRANT   DB 'Password accepted. Access granted.',0DH,0AH,0
LOCKMSG DB 'Too many failed attempts. SYSTEM LOCKED.',0DH,0AH,0

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX
    MOV AX, 0003H                     ; INT 10H / AH=00H : set 80x25 text mode (clears screen)
    INT 10H
    LEA SI, BANNER
    CALL PUTS
ATTEMPT:
    LEA SI, PROMPT
    CALL PUTS
    CALL GETPASS                      ; hidden input -> BUF
    CALL VERIFY                       ; AL = 1 if BUF equals PASSWD
    CMP AL, 1
    JE GRANTED
    DEC TRIES                         ; one attempt used
    JZ LOCKED                         ; no attempts left -> lock the system
    LEA SI, WRONG
    CALL PUTS
    MOV AL, TRIES
    ADD AL, '0'                       ; attempts left as an ASCII digit
    CALL PUTCH
    CALL NEWLINE
    JMP ATTEMPT
GRANTED:
    LEA SI, GRANT
    CALL PUTS
    MOV AX, 4C00H                     ; exit to DOS, code 0
    INT 21H
LOCKED:
    LEA SI, LOCKMSG
    CALL PUTS
    MOV AX, 4C01H                     ; exit to DOS, code 1 (locked)
    INT 21H
MAIN ENDP

; PUTCH: print character in AL using BIOS teletype (INT 10H / AH=0EH)
PUTCH PROC
    PUSH AX
    PUSH BX
    MOV AH, 0EH                       ; teletype output
    MOV BX, 0007H                     ; BH = page 0, BL = light grey
    INT 10H
    POP BX
    POP AX
    RET
PUTCH ENDP

; PUTS: print NULL terminated string at DS:SI through INT 10H
PUTS PROC
    PUSH AX
    PUSH SI
PS1:
    MOV AL, [SI]
    CMP AL, 0
    JE PS2
    CALL PUTCH
    INC SI
    JMP PS1
PS2:
    POP SI
    POP AX
    RET
PUTS ENDP

; NEWLINE: print CR + LF through INT 10H
NEWLINE PROC
    PUSH AX
    MOV AL, 0DH
    CALL PUTCH
    MOV AL, 0AH
    CALL PUTCH
    POP AX
    RET
NEWLINE ENDP

; GETPASS: read keys with INT 16H (AH=00H) WITHOUT echoing; Enter ends the input
GETPASS PROC
    PUSH AX
    PUSH CX
    PUSH DI
    LEA DI, BUF                       ; DI = next free byte in BUF
    XOR CX, CX                        ; CX = characters typed so far
GP1:
    MOV AH, 00H                       ; INT 16H / AH=00H : wait for a key, AL = ASCII code
    INT 16H
    CMP AL, 0DH                       ; Enter?
    JE GP4
    CMP AL, 08H                       ; Backspace?
    JE GP3
    CMP AL, 20H
    JB GP1                            ; ignore control / extended keys
    CMP CX, MAXLEN
    JAE GP1                           ; buffer full: ignore key
    MOV [DI], AL                      ; store key silently (no echo)
    INC DI
    INC CX
    JMP GP1
GP3:
    JCXZ GP1                          ; nothing to erase
    DEC DI
    DEC CX
    JMP GP1
GP4:
    MOV BYTE PTR [DI], 0              ; terminate the typed string
    CALL NEWLINE
    POP DI
    POP CX
    POP AX
    RET
GETPASS ENDP

; VERIFY: compare BUF with PASSWD; returns AL = 1 (match) or 0 (mismatch)
VERIFY PROC
    PUSH SI
    PUSH DI
    LEA SI, BUF
    LEA DI, PASSWD
V1:
    MOV AL, [SI]
    CMP AL, [DI]
    JNE VBAD                          ; first difference -> wrong password
    CMP AL, 0
    JE VOK                            ; both strings ended together -> equal
    INC SI
    INC DI
    JMP V1
VOK:
    MOV AL, 1
    JMP VEND
VBAD:
    MOV AL, 0
VEND:
    POP DI
    POP SI
    RET
VERIFY ENDP

END MAIN

Design and implement a simple command-line shell emulator using MASM (Microsoft Macro Assembler) that accepts user commands from the terminal and performs appropriate actions. The shell must interpret the following commands: (a) DIR – display a list of mock file names (simulate file listing), (b) TYPE <filename> – display the contents of the specified file, (c) COPY <source> <destination> – copy the contents of one file to another, (d) EXIT – exit the shell interface and terminate the program, (e) the emulator should prompt the user for input, parse the entered command, and invoke the appropriate function to operate. All actions should be implemented using MASM and DOS system interrupts (INT 21h) where applicable.

MASM Source Code
.MODEL SMALL
.STACK 100H

PRINT MACRO TXT                       ; print a NULL terminated string
    LEA SI, TXT
    CALL PUTS
ENDM

.DATA
FNAME_SZ EQU 13                       ; bytes per file-name slot (12 chars + 0)
FDATA_SZ EQU 80                       ; bytes per file-content slot
NFILES   EQU 5                        ; number of slots in the mock disk

; ---- mock disk kept in RAM: 5 name slots and 5 content slots ----
NAMES   DB 65 DUP(0)                  ; 5 x 13
CONTS   DB 400 DUP(0)                 ; 5 x 80
; ---- initial files (copied into the slots by INITDISK) ----
IN_N1   DB 'README.TXT',0
IN_C1   DB 'Welcome to the MASM shell emulator.',0
IN_N2   DB 'HELLO.ASM',0
IN_C2   DB 'MOV AH,09H / LEA DX,MSG / INT 21H',0
IN_N3   DB 'NOTES.DAT',0
IN_C3   DB 'Lab report due on Friday.',0

INBUF   DB 50,0,50 DUP(0)             ; INT 21H/0AH buffer: max, length, text
TOKS    DW 3 DUP(0)                   ; pointers to the 3 command tokens
TOKCNT  DB 0                          ; number of tokens found
SRCSLOT DB 0
DSTSLOT DB 0
NFOUND  DW 0                          ; files counted by DIR

BANNER  DB 'MASM Shell Emulator v1.0  (commands: DIR, TYPE, COPY, EXIT)',0DH,0AH,0
PROMPT  DB 'MASM-SHELL> ',0
HDR     DB ' Directory of A:\',0DH,0AH,0
BYTESM  DB ' bytes',0DH,0AH,0
FILESM  DB ' file(s)',0DH,0AH,0
BADCMD  DB 'Bad command or file name',0DH,0AH,0
NOFILE  DB 'File not found',0DH,0AH,0
USE_T   DB 'Usage: TYPE <filename>',0DH,0AH,0
USE_C   DB 'Usage: COPY <source> <destination>',0DH,0AH,0
FULLM   DB 'Disk full - no free directory entry',0DH,0AH,0
COPIED  DB '        1 file(s) copied',0DH,0AH,0
BYEM    DB 'Shell terminated. Goodbye!',0DH,0AH,0
CRLF    DB 0DH,0AH,0
C_DIR   DB 'DIR',0
C_TYPE  DB 'TYPE',0
C_COPY  DB 'COPY',0
C_EXIT  DB 'EXIT',0

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX
    CALL INITDISK                     ; build the mock files
    PRINT BANNER
SHELL:
    PRINT PROMPT
    LEA DX, INBUF
    MOV AH, 0AH                       ; INT 21H/0AH : buffered keyboard input
    INT 21H
    PRINT CRLF
    CALL PARSE                        ; upper-case the line and split it into tokens
    CMP TOKCNT, 0
    JE SHELL                          ; empty line -> prompt again

    MOV SI, TOKS                      ; SI = first token (the command word)
    LEA DI, C_DIR
    CALL STREQ
    CMP AL, 1
    JNE NOTDIR
    CALL CMD_DIR
    JMP SHELL
NOTDIR:
    MOV SI, TOKS
    LEA DI, C_TYPE
    CALL STREQ
    CMP AL, 1
    JNE NOTTYPE
    CALL CMD_TYPE
    JMP SHELL
NOTTYPE:
    MOV SI, TOKS
    LEA DI, C_COPY
    CALL STREQ
    CMP AL, 1
    JNE NOTCOPY
    CALL CMD_COPY
    JMP SHELL
NOTCOPY:
    MOV SI, TOKS
    LEA DI, C_EXIT
    CALL STREQ
    CMP AL, 1
    JE QUIT
    PRINT BADCMD                      ; unknown command
    JMP SHELL
QUIT:
    PRINT BYEM
    MOV AX, 4C00H                     ; INT 21H/4CH : terminate program
    INT 21H
MAIN ENDP

; INITDISK: copy the initial names/contents into slots 0, 1 and 2
INITDISK PROC
    LEA SI, IN_N1
    LEA DI, NAMES
    CALL STRCPY
    LEA SI, IN_C1
    LEA DI, CONTS
    CALL STRCPY
    LEA SI, IN_N2
    LEA DI, NAMES+FNAME_SZ
    CALL STRCPY
    LEA SI, IN_C2
    LEA DI, CONTS+FDATA_SZ
    CALL STRCPY
    LEA SI, IN_N3
    LEA DI, NAMES+2*FNAME_SZ
    CALL STRCPY
    LEA SI, IN_C3
    LEA DI, CONTS+2*FDATA_SZ
    CALL STRCPY
    RET
INITDISK ENDP

; PARSE: upper-case INBUF text and split it at spaces into TOKS[0..2]
PARSE PROC
    PUSH AX
    PUSH BX
    PUSH SI
    MOV BL, INBUF+1                   ; BL = number of characters typed
    XOR BH, BH
    MOV BYTE PTR INBUF[BX+2], 0       ; replace the CR by a 0 terminator
    MOV WORD PTR TOKS, 0              ; clear token table
    MOV WORD PTR TOKS+2, 0
    MOV WORD PTR TOKS+4, 0
    LEA SI, INBUF+2                   ; SI = first typed character
UP1:
    MOV AL, [SI]
    CMP AL, 0
    JE UP3
    CMP AL, 'a'
    JB UP2
    CMP AL, 'z'
    JA UP2
    SUB AL, 20H                       ; lower case -> upper case
    MOV [SI], AL
UP2:
    INC SI
    JMP UP1
UP3:
    LEA SI, INBUF+2
    XOR BX, BX                        ; BX = token index x 2
NEXTTOK:
SK:
    MOV AL, [SI]                      ; skip leading spaces
    CMP AL, ' '
    JNE SK2
    INC SI
    JMP SK
SK2:
    CMP AL, 0
    JE PDONE                          ; end of line
    CMP BX, 6
    JAE PDONE                         ; only 3 tokens are kept
    MOV TOKS[BX], SI                  ; remember where this token starts
    ADD BX, 2
SC:
    MOV AL, [SI]                      ; scan to the end of the token
    CMP AL, 0
    JE PDONE
    CMP AL, ' '
    JE SC2
    INC SI
    JMP SC
SC2:
    MOV BYTE PTR [SI], 0              ; end the token with 0
    INC SI
    JMP NEXTTOK
PDONE:
    SHR BX, 1                         ; BX = number of tokens
    MOV TOKCNT, BL
    POP SI
    POP BX
    POP AX
    RET
PARSE ENDP

; ---------- DIR : list every used slot with its size ----------
CMD_DIR PROC
    PUSH AX
    PUSH BX
    PRINT HDR
    MOV NFOUND, 0
    XOR BX, BX                        ; BX = slot number
DL1:
    CALL SHOWENTRY
    INC BX
    CMP BX, NFILES
    JB DL1
    MOV AX, NFOUND
    CALL PRINT_NUM
    PRINT FILESM
    POP BX
    POP AX
    RET
CMD_DIR ENDP

; SHOWENTRY: BX = slot; prints "NAME   size bytes" if the slot is used
SHOWENTRY PROC
    PUSH AX
    PUSH CX
    PUSH DX
    PUSH SI
    PUSH DI
    MOV AX, BX
    CALL NAMEADDR                     ; DI = address of the name
    CMP BYTE PTR [DI], 0
    JE SEDONE                         ; free slot
    MOV SI, DI
    CALL PUTS                         ; print the file name
    CALL STRLEN                       ; AX = name length
    MOV CX, 14
    SUB CX, AX                        ; CX = padding spaces
SEPAD:
    MOV DL, ' '
    MOV AH, 02H
    INT 21H
    LOOP SEPAD
    MOV AX, BX
    CALL CONTADDR                     ; DI = address of the content
    MOV SI, DI
    CALL STRLEN                       ; AX = file size in bytes
    CALL PRINT_NUM
    PRINT BYTESM
    INC NFOUND
SEDONE:
    POP DI
    POP SI
    POP DX
    POP CX
    POP AX
    RET
SHOWENTRY ENDP

; ---------- TYPE <filename> ----------
CMD_TYPE PROC
    CMP TOKCNT, 2
    JAE TY1
    PRINT USE_T
    RET
TY1:
    MOV SI, TOKS+2                    ; SI = second token (file name)
    CALL FINDFILE                     ; AL = slot or 0FFH
    CMP AL, 0FFH
    JNE TY2
    PRINT NOFILE
    RET
TY2:
    CALL CONTADDR                     ; DI = file content
    MOV SI, DI
    CALL PUTS
    PRINT CRLF
    RET
CMD_TYPE ENDP

; ---------- COPY <source> <destination> ----------
CMD_COPY PROC
    CMP TOKCNT, 3
    JAE CP_OK
    PRINT USE_C
    RET
CP_OK:
    MOV SI, TOKS+2                    ; source name
    CALL FINDFILE
    CMP AL, 0FFH
    JNE CP1
    PRINT NOFILE
    RET
CP1:
    MOV SRCSLOT, AL
    MOV SI, TOKS+4                    ; destination name
    CALL FINDFILE
    CMP AL, 0FFH
    JE CP2                            ; destination does not exist -> create it
    MOV DSTSLOT, AL                   ; destination exists -> overwrite its content
    JMP CP5
CP2:
    XOR BX, BX                        ; search for a free slot
CP2L:
    MOV AX, BX
    CALL NAMEADDR                     ; DI = name field of slot BX
    CMP BYTE PTR [DI], 0
    JE CP3
    INC BX
    CMP BX, NFILES
    JB CP2L
    PRINT FULLM                       ; no free slot
    RET
CP3:
    MOV DSTSLOT, BL
    MOV SI, TOKS+4
    MOV CX, 12                        ; keep at most 12 characters of the name
CP3L:
    MOV AL, [SI]
    MOV [DI], AL
    CMP AL, 0
    JE CP5                            ; whole name copied (0 included)
    INC SI
    INC DI
    LOOP CP3L
    MOV BYTE PTR [DI], 0              ; name was longer than 12: cut it
CP5:
    MOV AL, SRCSLOT
    CALL CONTADDR                     ; DI = source content
    MOV SI, DI
    MOV AL, DSTSLOT
    CALL CONTADDR                     ; DI = destination content
CP6:
    MOV AL, [SI]                      ; copy content including the 0
    MOV [DI], AL
    INC SI
    INC DI
    CMP AL, 0
    JNE CP6
    PRINT COPIED
    RET
CMD_COPY ENDP

; FINDFILE: SI = file name; returns AL = slot number or 0FFH if not found
FINDFILE PROC
    PUSH BX
    PUSH DI
    XOR BX, BX
FF1:
    MOV AX, BX
    CALL NAMEADDR                     ; DI = name of slot BX
    CMP BYTE PTR [DI], 0
    JE FF2                            ; free slot: skip
    CALL STREQ                        ; compare SI with DI
    CMP AL, 1
    JE FFOK
FF2:
    INC BX
    CMP BX, NFILES
    JB FF1
    MOV AL, 0FFH                      ; not found
    JMP FFEND
FFOK:
    MOV AX, BX                        ; AL = slot number
FFEND:
    POP DI
    POP BX
    RET
FINDFILE ENDP

; NAMEADDR: AL = slot -> DI = NAMES + slot*13
NAMEADDR PROC
    PUSH AX
    PUSH BX
    MOV BL, FNAME_SZ
    MUL BL                            ; AX = AL * 13
    LEA DI, NAMES
    ADD DI, AX
    POP BX
    POP AX
    RET
NAMEADDR ENDP

; CONTADDR: AL = slot -> DI = CONTS + slot*80
CONTADDR PROC
    PUSH AX
    PUSH BX
    MOV BL, FDATA_SZ
    MUL BL                            ; AX = AL * 80
    LEA DI, CONTS
    ADD DI, AX
    POP BX
    POP AX
    RET
CONTADDR ENDP

; STREQ: compare strings at SI and DI; AL = 1 if equal, 0 otherwise
STREQ PROC
    PUSH SI
    PUSH DI
EQ1:
    MOV AL, [SI]
    CMP AL, [DI]
    JNE EQ_NO
    OR AL, AL
    JZ EQ_YES                         ; both ended -> equal
    INC SI
    INC DI
    JMP EQ1
EQ_YES:
    MOV AL, 1
    JMP EQ_END
EQ_NO:
    XOR AL, AL
EQ_END:
    POP DI
    POP SI
    RET
STREQ ENDP

; STRLEN: SI = string; returns AX = length (SI preserved)
STRLEN PROC
    PUSH SI
    XOR AX, AX
SL1:
    CMP BYTE PTR [SI], 0
    JE SL2
    INC SI
    INC AX
    JMP SL1
SL2:
    POP SI
    RET
STRLEN ENDP

; STRCPY: copy NULL terminated string from SI to DI
STRCPY PROC
    PUSH AX
    PUSH SI
    PUSH DI
SCP1:
    MOV AL, [SI]
    MOV [DI], AL
    INC SI
    INC DI
    CMP AL, 0
    JNE SCP1
    POP DI
    POP SI
    POP AX
    RET
STRCPY ENDP

; PUTS: print NULL terminated string at SI using INT 21H/02H
PUTS PROC
    PUSH AX
    PUSH DX
    PUSH SI
PU1:
    MOV DL, [SI]
    CMP DL, 0
    JE PU2
    MOV AH, 02H
    INT 21H
    INC SI
    JMP PU1
PU2:
    POP SI
    POP DX
    POP AX
    RET
PUTS ENDP

; PRINT_NUM: print unsigned number in AX (all registers preserved)
PRINT_NUM PROC
    PUSH AX
    PUSH BX
    PUSH CX
    PUSH DX
    MOV BX, 10
    XOR CX, CX
PN1:
    XOR DX, DX
    DIV BX
    PUSH DX
    INC CX
    OR AX, AX
    JNZ PN1
PN2:
    POP DX
    ADD DL, '0'
    MOV AH, 02H
    INT 21H
    LOOP PN2
    POP DX
    POP CX
    POP BX
    POP AX
    RET
PRINT_NUM ENDP

END MAIN
